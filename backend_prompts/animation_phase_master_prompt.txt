def show_solution(self):
    active_mobs = list(self._active_from_question)

    fig = self._fig_group
    self.play(fig.animate.move_to(RIGHT * 3.0), run_time=1.0)

    badge = create_heading_badge("Solution")
    badge_old = active_mobs[0]
    self.play(FadeOut(badge_old), FadeIn(badge), run_time=0.5)
    active_mobs[0] = badge

    texts_to_clear = [
        m for m in active_mobs
        if m is not badge and m is not self._fig_group
    ]
    if texts_to_clear:
        self.play(*[FadeOut(t) for t in texts_to_clear], run_time=0.5)
        for t in texts_to_clear:
            active_mobs.remove(t)
    self.wait(0.2)

    with self.voiceover(text='...') as tracker:
        mgr = StepManager(
            self,
            start_anchor=UP*2.0 + LEFT*3.5,
            font_size=28,
            buff=0.3
        )

        self.wait_until_bookmark("bk_s1")
        s1 = math_obj(r"A = l \times w", font_size=28)
        mgr.add_step(s1)
        active_mobs.append(s1)

        self.wait_until_bookmark("bk_s2")
        s2 = math_obj(r"= 10 \times 5", font_size=28)
        mgr.add_step(s2)
        active_mobs.append(s2)

        self.wait_until_bookmark("bk_s3")
        s3 = math_obj(r"= 50 \text{ cm}^2", font_size=28)
        s3.set_color(ORANGE_HL)
        mgr.add_step(s3)
        active_mobs.append(s3)

    self.wait(0.6)
    clear_and_transition(self, active_mobs, LAVENDER_BG)
    active_mobs.clear()
